            const newY = bot.y + Math.sin(angle) * dist;

           

            if (!WorldEngine.isBlocked(newX, newY)) {

              bot.targetX = newX;

              bot.targetY = newY;

            }

          const dy = bot.targetY - bot.y;

            bot.x += (dx / dist) * speed * dt;

            bot.y += (dy / dist) * speed * dt;

            bot.animFrame = (Math.floor(Date.now() / 150)) % 4;

            if (Math.abs(dx) > Math.abs(dy)) {

              bot.direction = dx > 0 ? 'right' : 'left';

              bot.direction = dy > 0 ? 'down' : 'up';

          }

      updateJoystick(clientX, clientY, knob) {

        let diffX = clientX - this.joystick.startX;

        let diffY = clientY - this.joystick.startY;

        const dist = Math.hypot(diffX, diffY);

        if (dist > this.joystick.maxRadius) {

          diffX = (diffX / dist) * this.joystick.maxRadius;

          diffY = (diffY / dist) * this.joystick.maxRadius;

        }

        knob.style.transform = `translate(${diffX}px, ${diffY}px)`;

        this.joystick.dx = diffX / this.joystick.maxRadius;

        this.joystick.dy = diffY / this.joystick.maxRadius;

      updatePlayerMovement(dt) {

        let moveX = 0;

        let moveY = 0;

        // Input Keyboard WASD & Panah

        if (GameState.keys['w'] || GameState.keys['arrowup']) moveY -= 1;

        if (GameState.keys['s'] || GameState.keys['arrowdown']) moveY += 1;

        if (GameState.keys['a'] || GameState.keys['arrowleft']) moveX -= 1;

        if (GameState.keys['d'] || GameState.keys['arrowright']) moveX += 1;

        // Input Virtual Joystick

        if (this.joystick.active) {

          moveX = this.joystick.dx;

          moveY = this.joystick.dy;

        }

        const len = Math.hypot(moveX, moveY);

        if (len > 0.15) {

          Player.isMoving = true;

          const normX = moveX / (len > 1 ? len : 1);

          const normY = moveY / (len > 1 ? len : 1);

          const nextX = Player.x + normX * Player.speed * dt;

          const nextY = Player.y + normY * Player.speed * dt;

          // Cek tabrakan sumbu X

          if (!WorldEngine.isBlocked(nextX, Player.y)) {

            Player.x = nextX;

          }

            Player.direction = normY > 0 ? 'down' : 'up';

          }

          Player.animFrame = Math.floor(Player.animTimer) % 4;

            SoundFX.playWalk();

          Player.isMoving = false;

          Player.animFrame = 0;

        }

        // Update Attack Cooldown

        if (Player.attackCooldown > 0) Player.attackCooldown -= dt;

        // Update Chat Bubble Timer

        if (Player.chatTimer > 0) {

          Player.chatTimer -= dt;

          if (Player.chatTimer <= 0) Player.chatBubble = null;

        }

      }
